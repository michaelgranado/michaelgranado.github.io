__global__ void kernelRenderPixels() {
    // Calculate pixel coordinates
    int imageX = blockIdx.x * blockDim.x + threadIdx.x;
    int imageY = blockIdx.y * blockDim.y + threadIdx.y;
    
    // Get image dimensions
    int imageWidth = cuConstRendererParams.imageWidth;
    int imageHeight = cuConstRendererParams.imageHeight;
    
    // Check if pixel is in bounds
    if (imageX >= imageWidth || imageY >= imageHeight)
        return;
    
    // Calculate pixel center in normalized coordinates
    float invWidth = 1.0f / imageWidth;
    float invHeight = 1.0f / imageHeight;
    float2 pixelCenter = make_float2(invWidth * (static_cast(imageX) + 0.5f),
                                     invHeight * (static_cast(imageY) + 0.5f));
    
    // Calculate pixel offset in image array
    int offset = 4 * (imageY * imageWidth + imageX);
    float4 pixelColor = make_float4(cuConstRendererParams.imageData[offset],
                                    cuConstRendererParams.imageData[offset + 1],
                                    cuConstRendererParams.imageData[offset + 2],
                                    cuConstRendererParams.imageData[offset + 3]);
    
    // Process all circles in order
    for (int i = 0; i < cuConstRendererParams.numCircles; i++) {
        int index3 = 3 * i;
        float3 p = *(float3*)(&cuConstRendererParams.position[index3]);
        float rad = cuConstRendererParams.radius[i];
        
        // Check if circle covers pixel
        float diffX = p.x - pixelCenter.x;
        float diffY = p.y - pixelCenter.y;
        float pixelDist = diffX * diffX + diffY * diffY;
        float maxDist = rad * rad;
        
        if (pixelDist <= maxDist) {
            // Circle covers pixel, apply blend function
            float3 rgb;
            float alpha;
            
            // Color calculation (similar to shadePixel)
            if (cuConstRendererParams.sceneName == SNOWFLAKES || cuConstRendererParams.sceneName == SNOWFLAKES_SINGLE_FRAME) {
                const float kCircleMaxAlpha = .5f;
                const float falloffScale = 4.f;
                
                float normPixelDist = sqrt(pixelDist) / rad;
                rgb = lookupColor(normPixelDist);
                
                float maxAlpha = .6f + .4f * (1.f-p.z);
                maxAlpha = kCircleMaxAlpha * fmaxf(fminf(maxAlpha, 1.f), 0.f);
                alpha = maxAlpha * exp(-1.f * falloffScale * normPixelDist * normPixelDist);
            } else {
                rgb = *(float3*)&(cuConstRendererParams.color[index3]);
                alpha = .5f;
            }
            
            // Blend function
            float oneMinusAlpha = 1.f - alpha;
            pixelColor.x = alpha * rgb.x + oneMinusAlpha * pixelColor.x;
            pixelColor.y = alpha * rgb.y + oneMinusAlpha * pixelColor.y;
            pixelColor.z = alpha * rgb.z + oneMinusAlpha * pixelColor.z;
            pixelColor.w = alpha + pixelColor.w;
        }
    }
    
    // Write final color to output
    cuConstRendererParams.imageData[offset] = pixelColor.x;
    cuConstRendererParams.imageData[offset + 1] = pixelColor.y;
    cuConstRendererParams.imageData[offset + 2] = pixelColor.z;
    cuConstRendererParams.imageData[offset + 3] = pixelColor.w;
}
