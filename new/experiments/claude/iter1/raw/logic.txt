<h2 style="color: #333;">CUDA Renderer Correction</h2>

<div style="background-color: #f8f9fa; padding: 15px; margin: 20px 0; border-radius: 5px;">
<h3 style="margin-top: 0;">Problem Description</h3>
<p>The current CUDA renderer implementation violates both the atomicity and order requirements. Multiple threads can update the same pixel simultaneously, leading to race conditions and incorrect rendering. This is evident from the horizontal streaks in the resulting images.</p>
</div>

<div style="background-color: #f8f9fa; padding: 15px; margin: 20px 0; border-radius: 5px;">
<h3 style="margin-top: 0;">Problem Reasoning</h3>
<p>The current implementation assigns one thread per circle, which is problematic for the following reasons:</p>
<ol>
    <li>When multiple circles overlap at a pixel, multiple threads might try to update the same pixel simultaneously, causing race conditions.</li>
    <li>There's no guarantee that the updates are applied in the correct order, which is essential for proper transparency rendering.</li>
    <li>The read-modify-write operations in <code>shadePixel</code> function are not atomic, as indicated by the comments "BEGIN SHOULD-BE-ATOMIC REGION" and "END SHOULD-BE-ATOMIC REGION".</li>
</ol>
</div>

<div style="background-color: #f8f9fa; padding: 15px; margin: 20px 0; border-radius: 5px;">
<h3 style="margin-top: 0;">Proposed Remedy</h3>
<p>To fix these issues, I propose changing the parallelization strategy from "one thread per circle" to "one thread per pixel". Each thread will be responsible for one pixel, process all circles in order, and only write to the image once at the end.</p>
</div>

<div style="background-color: #f8f9fa; padding: 15px; margin: 20px 0; border-radius: 5px;">
<h3 style="margin-top: 0;">Remedy Reasoning</h3>
<p>This solution addresses the two core requirements:</p>
<ol>
    <li><strong>Atomicity:</strong> By having each thread process just one pixel, we ensure atomicity since there's no concurrent access to the same pixel.</li>
    <li><strong>Order:</strong> By processing circles in the order they are given, we ensure the order requirement is met.</li>
</ol>
<p>1) Why will this fix the problem? Each pixel is processed by exactly one thread, eliminating race conditions. The circles are processed in order, ensuring correct transparency blending.</p>
<p>2) Expected results: The renderer will produce correct images without any horizontal streaks or artifacts for all test cases, including scenes with transparent circles.</p>
</div>

// New kernel that renders all pixels with one thread per pixel
__global__ void kernelRenderPixels() {
    int pixelX = blockIdx.x * blockDim.x + threadIdx.x;
    int pixelY = blockIdx.y * blockDim.y + threadIdx.y;
    
    // Early exit for threads that are outside the image bounds
    if (pixelX >= cuConstRendererParams.imageWidth || pixelY >= cuConstRendererParams.imageHeight)
        return;
        
    // Initialize the pixel color based on the scene
    float4 pixelColor;
    if (cuConstRendererParams.sceneName == SNOWFLAKES || cuConstRendererParams.sceneName == SNOWFLAKES_SINGLE_FRAME) {
        float shade = .4f + .45f * static_cast<float>(cuConstRendererParams.imageHeight-pixelY) / cuConstRendererParams.imageHeight;
        pixelColor = make_float4(shade, shade, shade, 1.f);
    } else {
        pixelColor = make_float4(1.f, 1.f, 1.f, 1.f);
    }
    
    // Compute the normalized pixel center
    float invWidth = 1.f / cuConstRendererParams.imageWidth;
    float invHeight = 1.f / cuConstRendererParams.imageHeight;
    float2 pixelCenterNorm = make_float2(
        invWidth * (static_cast<float>(pixelX) + 0.5f),
        invHeight * (static_cast<float>(pixelY) + 0.5f)
    );
    
    // Process all circles in the order they are given
    for (int circleIndex = 0; circleIndex < cuConstRendererParams.numCircles; circleIndex++) {
        int index3 = 3 * circleIndex;
        float3 p = *(float3*)(&cuConstRendererParams.position[index3]);
        float rad = cuConstRendererParams.radius[circleIndex];
        
        // Check if this pixel is within the circle
        float diffX = p.x - pixelCenterNorm.x;
        float diffY = p.y - pixelCenterNorm.y;
        float pixelDist = diffX * diffX + diffY * diffY;
        
        if (pixelDist <= rad * rad) {
            float3 rgb;
            float alpha;
            
            // Compute the shading value
            if (cuConstRendererParams.sceneName == SNOWFLAKES || cuConstRendererParams.sceneName == SNOWFLAKES_SINGLE_FRAME) {
                const float kCircleMaxAlpha = .5f;
                const float falloffScale = 4.f;
                
                float normPixelDist = sqrt(pixelDist) / rad;
                rgb = lookupColor(normPixelDist);
                
                float maxAlpha = .6f + .4f * (1.f-p.z);
                maxAlpha = kCircleMaxAlpha * fmaxf(fminf(maxAlpha, 1.f), 0.f);
                alpha = maxAlpha * exp(-1.f * falloffScale * normPixelDist * normPixelDist);
            } else {
                rgb = *(float3*)&(cuConstRendererParams.color[index3]);
                alpha = .5f;
            }
            
            // Blend the contribution with the current pixel color
            float oneMinusAlpha = 1.f - alpha;
            pixelColor.x = alpha * rgb.x + oneMinusAlpha * pixelColor.x;
            pixelColor.y = alpha * rgb.y + oneMinusAlpha * pixelColor.y;
            pixelColor.z = alpha * rgb.z + oneMinusAlpha * pixelColor.z;
            pixelColor.w = alpha + pixelColor.w;
        }
    }
    
    // Write the final pixel color to the image
    int offset = 4 * (pixelY * cuConstRendererParams.imageWidth + pixelX);
    *(float4*)(&cuConstRendererParams.imageData[offset]) = pixelColor;
}

// Update the render method to use the new kernel
void CudaRenderer::render() {
    // Set up the grid dimensions for a "one thread per pixel" approach
    dim3 blockDim(16, 16);
    dim3 gridDim(
        (image->width + blockDim.x - 1) / blockDim.x,
        (image->height + blockDim.y - 1) / blockDim.y);
    
    kernelRenderPixels<<<gridDim, blockDim>>>();
    cudaDeviceSynchronize();
}

The semi-transparent circle rendering requires that circles be processed in a specific order for proper blending. When multiple threads update the same pixel without coordination, we get the horizontal streaks visible in the buggy output images. These issues violate both the atomicity and ordering requirements specified in the assignment.</p>

Each thread will be responsible for processing all circles that affect its assigned pixel, in the order the circles were provided. This ensures both atomicity (since only one thread updates each pixel) and correct ordering (since each thread processes circles sequentially).</p>

// kernelRenderCirclesOrdered -- (CUDA device code)
//
// Each thread renders a single pixel, processing all circles in order
__global__ void kernelRenderCirclesOrdered() {
    int pixelX = blockIdx.x * blockDim.x + threadIdx.x;
    int pixelY = blockIdx.y * blockDim.y + threadIdx.y;
    
    int imageWidth = cuConstRendererParams.imageWidth;
    int imageHeight = cuConstRendererParams.imageHeight;
    
    // Check if this thread's pixel is within the image bounds
    if (pixelX >= imageWidth || pixelY >= imageHeight)
        return;
    
    float invWidth = 1.f / imageWidth;
    float invHeight = 1.f / imageHeight;
    float2 pixelCenterNorm = make_float2(invWidth * (static_cast<float>(pixelX) + 0.5f),
                                        invHeight * (static_cast<float>(pixelY) + 0.5f));
    
    // Get a pointer to this pixel's color data
    float4* imgPtr = (float4*)(&cuConstRendererParams.imageData[4 * (pixelY * imageWidth + pixelX)]);
    
    // Process all circles in order
    for (int circleIndex = 0; circleIndex < cuConstRendererParams.numCircles; circleIndex++) {
        int index3 = 3 * circleIndex;
        
        // Read circle position and radius
        float3 p = *(float3*)(&cuConstRendererParams.position[index3]);
        float rad = cuConstRendererParams.radius[circleIndex];
        
        // Compute the distance from pixel center to circle center
        float diffX = p.x - pixelCenterNorm.x;
        float diffY = p.y - pixelCenterNorm.y;
        float pixelDist = diffX * diffX + diffY * diffY;
        float maxDist = rad * rad;
        
        // If pixel is within the circle, update its color
        if (pixelDist <= maxDist) {
            shadePixel(circleIndex, pixelCenterNorm, p, imgPtr);
        }
    }
}

void
CudaRenderer::render() {
    // Set up a 2D grid of threads, where each thread processes one pixel
    dim3 blockDim(16, 16);
    dim3 gridDim(
        (image->width + blockDim.x - 1) / blockDim.x,
        (image->height + blockDim.y - 1) / blockDim.y);
    
    kernelRenderCirclesOrdered<<<gridDim, blockDim>>>();
    cudaDeviceSynchronize();
}
