CUDA Renderer Optimization

Problem Description

The current implementation processes each pixel independently through all circles in the scene, which is inefficient. For each pixel, every circle is tested regardless of whether it could potentially affect that pixel. This causes excessive computation, especially with scenes containing many circles (like rand1M and micro2M), leading to poor performance.

Problem Reasoning

Upon analyzing the current implementation, I noticed that while it guarantees correctness by processing pixels in parallel and maintaining the required order of circle rendering, it's computationally inefficient. Each thread processes every circle for its assigned pixel, even those that can't possibly intersect with it. This redundant work significantly slows down performance for scenes with many circles. Additionally, there's little data locality, leading to suboptimal memory access patterns and cache utilization.

Proposed Remedy

Implement a tile-based approach where the image is divided into tiles (blocks of pixels), and each tile is processed by a thread block. For each tile, determine which circles might affect it using a conservative circle-box test. Then, only process those relevant circles for the pixels in that tile. Use shared memory to cache circle data and test results, further improving performance by reducing global memory access.

Remedy Reasoning

This approach maintains correctness while significantly improving performance by:

  1. Reducing unnecessary work: By quickly rejecting circles that don't affect a tile, we avoid many expensive pixel-circle tests.
  2. Improving memory access patterns: Caching circle data in shared memory reduces global memory access and improves cache utilization.
  3. Maintaining correctness: By processing circles in their original order and accumulating color locally, we preserve both the ordering and atomicity requirements.

1) This solution fixes the problem by eliminating redundant calculations while preserving the required rendering order. The circle-box test lets us quickly determine if a circle might affect any pixel in a tile.

2) I expect the program to produce identical results to the reference implementation, but with significantly improved performance, especially for scenes with many circles or where circles are unevenly distributed. The performance gains should be most noticeable in rand1M and micro2M tests.

Code

#define TILE_SIZE 16
#define CIRCLE_BATCH_SIZE 64  // Adjust based on available shared memory

__global__ void kernelRenderCirclesTiledOptimized() {
    // Tile coordinates
    int tileX = blockIdx.x;
    int tileY = blockIdx.y;
    
    // Pixel coordinates within the tile
    int pixelX = tileX * TILE_SIZE + threadIdx.x;
    int pixelY = tileY * TILE_SIZE + threadIdx.y;
    
    int imageWidth = cuConstRendererParams.imageWidth;
    int imageHeight = cuConstRendererParams.imageHeight;
    
    // Skip this thread if the pixel is outside the image bounds
    bool validPixel = (pixelX < imageWidth && pixelY < imageHeight);
    
    // Calculate normalized pixel center (only for valid pixels)
    float2 pixelCenterNorm;
    float4* imgPtr = nullptr;
    float4 pixelColor;
    
    if (validPixel) {
        float invWidth = 1.f / imageWidth;
        float invHeight = 1.f / imageHeight;
        pixelCenterNorm = make_float2(
            invWidth * (pixelX + 0.5f),
            invHeight * (pixelY + 0.5f)
        );
        
        // Get pointer to this pixel's color data
        imgPtr = (float4*)(&cuConstRendererParams.imageData[4 * (pixelY * imageWidth + pixelX)]);
        
        // Initialize local accumulator for the pixel color
        pixelColor = *imgPtr;
    }
    
    // Define the conservative bounding box for this tile
    float invWidth = 1.f / imageWidth;
    float invHeight = 1.f / imageHeight;
    float boxL = invWidth * (tileX * TILE_SIZE);
    float boxR = invWidth * min((tileX + 1) * TILE_SIZE, imageWidth);
    float boxB = invHeight * (tileY * TILE_SIZE);
    float boxT = invHeight * min((tileY + 1) * TILE_SIZE, imageHeight);
    
    // Shared memory for circle batch data and tile-circle test results
    __shared__ float3 sPositions[CIRCLE_BATCH_SIZE];
    __shared__ float sRadii[CIRCLE_BATCH_SIZE];
    __shared__ float3 sColors[CIRCLE_BATCH_SIZE];
    __shared__ bool sAffectsTile[CIRCLE_BATCH_SIZE];
    
    // Local thread index for shared memory loading
    int localThreadIdx = threadIdx.y * TILE_SIZE + threadIdx.x;
    int threadsPerBlock = TILE_SIZE * TILE_SIZE;
    
    // Process circles in batches
    for (int batchStart = 0; batchStart < cuConstRendererParams.numCircles; batchStart += CIRCLE_BATCH_SIZE) {
        int batchSize = min(CIRCLE_BATCH_SIZE, cuConstRendererParams.numCircles - batchStart);
        
        // Load circle data into shared memory and check tile-circle intersections
        for (int i = localThreadIdx; i < batchSize; i += threadsPerBlock) {
            int circleIndex = batchStart + i;
            int index3 = 3 * circleIndex;
            
            // Load position
            sPositions[i] = *(float3*)(&cuConstRendererParams.position[index3]);
            
            // Load radius
            sRadii[i] = cuConstRendererParams.radius[circleIndex];
            
            // Check if this circle affects this tile
            sAffectsTile[i] = circleInBoxConservative(
                sPositions[i].x, sPositions[i].y, sRadii[i],
                boxL, boxR, boxT, boxB
            );
            
            // Load color (if needed later)
            if (sAffectsTile[i]) {
                sColors[i] = *(float3*)(&cuConstRendererParams.color[index3]);
            }
        }
        
        __syncthreads();  // Ensure all threads have loaded shared memory and performed tests
        
        // Skip processing if this thread's pixel is outside the image bounds
        if (!validPixel) {
            __syncthreads();  // Necessary before next batch
            continue;
        }
        
        // Process all circles in this batch that affect this tile
        for (int i = 0; i < batchSize; i++) {
            if (!sAffectsTile[i]) {
                continue;  // Skip this circle for all pixels in this tile
            }
            
            float3 p = sPositions[i];
            float rad = sRadii[i];
            
            // Check if this circle affects this specific pixel
            float diffX = p.x - pixelCenterNorm.x;
            float diffY = p.y - pixelCenterNorm.y;
            float pixelDist = diffX * diffX + diffY * diffY;
            float maxDist = rad * rad;
            
            if (pixelDist <= maxDist) {
                // Compute the color contribution
                float3 rgb;
                float alpha;
                
                if (cuConstRendererParams.sceneName == SNOWFLAKES || cuConstRendererParams.sceneName == SNOWFLAKES_SINGLE_FRAME) {
                    const float kCircleMaxAlpha = .5f;
                    const float falloffScale = 4.f;
                    
                    float normPixelDist = sqrt(pixelDist) / rad;
                    rgb = lookupColor(normPixelDist);
                    
                    float maxAlpha = .6f + .4f * (1.f-p.z);
                    maxAlpha = kCircleMaxAlpha * fmaxf(fminf(maxAlpha, 1.f), 0.f);
                    alpha = maxAlpha * exp(-1.f * falloffScale * normPixelDist * normPixelDist);
                } else {
                    rgb = sColors[i];
                    alpha = .5f;
                }
                
                // Blend the color
                float oneMinusAlpha = 1.f - alpha;
                pixelColor.x = alpha * rgb.x + oneMinusAlpha * pixelColor.x;
                pixelColor.y = alpha * rgb.y + oneMinusAlpha * pixelColor.y;
                pixelColor.z = alpha * rgb.z + oneMinusAlpha * pixelColor.z;
                pixelColor.w = alpha + pixelColor.w;
            }
        }
        
        __syncthreads();  // Ensure all threads are done with shared memory before loading next batch
    }
    
    // Write the final color back to global memory (only once per pixel)
    if (validPixel) {
        *imgPtr = pixelColor;
    }
}

void
CudaRenderer::render() {
    // Set up a 2D grid of threads, where each block is a tile
    dim3 blockDim(TILE_SIZE, TILE_SIZE);
    dim3 gridDim(
        (image->width + blockDim.x - 1) / blockDim.x,
        (image->height + blockDim.y - 1) / blockDim.y);

    kernelRenderCirclesTiledOptimized<<<gridDim, blockDim>>>();
    cudaDeviceSynchronize();
}

Circle Renderer Optimization

Problem Description:

The current implementation of the CUDA renderer performs redundant circle intersection tests for every pixel, checking all circles against every pixel. This becomes extremely inefficient for scenes with many circles (like the micro2M test case with 2 million circles), as most circles have small radii and don't overlap with most pixels.

Problem Reasoning:

After analyzing the current kernelRenderCirclesOrdered implementation, I noticed that each thread (representing a pixel) loops through all circles in the scene, performing an overlap test for each one. For scenes with millions of circles like micro2M, this means each pixel thread is doing millions of intersection tests, most of which immediately fail since circles only overlap a small portion of the image. These redundant tests waste GPU compute resources.

The current implementation has a good conservative box test to quickly reject circles, but it's still applied to every circle for every pixel, which becomes a bottleneck when there are millions of circles. If we could first filter out circles at a coarser granularity (tile level) before checking at the pixel level, we could significantly reduce the number of tests performed.

Proposed Remedy:

Implement a tile-based rendering approach where we first test if a circle overlaps with a tile (block of pixels) before testing individual pixels within that tile. This adds a hierarchical filtering step:

  1. First check if a circle overlaps with the current thread block's tile
  2. Only if it does, proceed to check if it overlaps with the specific pixel

This maintains the correct ordering of circle rendering while reducing the number of unnecessary intersection tests performed.

Remedy Reasoning:

This optimization exploits spatial locality - circles with small radii only affect a small portion of the screen. By adding a tile-based filtering step, we can quickly reject circles that don't overlap with a whole region of pixels, saving many redundant tests.

1) The solution will fix the problem by reducing the number of circle-pixel intersection tests. Each thread will only need to consider circles that potentially overlap with its tile, rather than all circles.

2) I expect the program to still produce correct results because:

  • We're still maintaining the proper ordering of circle processing
  • The conservative box test ensures we don't miss any circles that might contribute to a pixel
  • Each pixel still accumulates color contributions from all overlapping circles in the correct order
  • We're just adding an earlier filtering step to avoid unnecessary work

This approach should significantly improve performance for scenes with many small circles while maintaining correctness across all test cases. The improvement should be most dramatic for the micro2M test case.

Code:

__global__ void kernelRenderCirclesTiled() {
    // Get the global pixel index for this thread
    int pixelX = blockIdx.x * blockDim.x + threadIdx.x;
    int pixelY = blockIdx.y * blockDim.y + threadIdx.y;

    int imageWidth = cuConstRendererParams.imageWidth;
    int imageHeight = cuConstRendererParams.imageHeight;

    // Check if this thread's pixel is within the image bounds
    if (pixelX >= imageWidth || pixelY >= imageHeight)
        return;

    // Compute the normalized pixel center
    float invWidth = 1.f / imageWidth;
    float invHeight = 1.f / imageHeight;
    float2 pixelCenterNorm = make_float2(invWidth * (static_cast<float>(pixelX) + 0.5f),
                                        invHeight * (static_cast<float>(pixelY) + 0.5f));

    // Get a pointer to this pixel's color data
    float4* imgPtr = (float4*)(&cuConstRendererParams.imageData[4 * (pixelY * imageWidth + pixelX)]);

    // Use a local accumulator for the pixel color
    float4 pixelColor = *imgPtr;

    // Define the pixel's box for conservative circle intersection test
    float halfInvWidth = 0.5f * invWidth;
    float halfInvHeight = 0.5f * invHeight;
    float boxL = pixelCenterNorm.x - halfInvWidth;
    float boxR = pixelCenterNorm.x + halfInvWidth;
    float boxB = pixelCenterNorm.y - halfInvHeight;
    float boxT = pixelCenterNorm.y + halfInvHeight;

    // Define the tile's box for the entire thread block
    float tileL = invWidth * (blockIdx.x * blockDim.x);
    float tileR = invWidth * ((blockIdx.x + 1) * blockDim.x);
    float tileB = invHeight * (blockIdx.y * blockDim.y);
    float tileT = invHeight * ((blockIdx.y + 1) * blockDim.y);

    // Process all circles in order (maintaining the order invariant)
    for (int circleIdx = 0; circleIdx < cuConstRendererParams.numCircles; circleIdx++) {
        int index3 = 3 * circleIdx;

        // Read circle position and radius
        float3 p = *(float3*)(&cuConstRendererParams.position[index3]);
        float rad = cuConstRendererParams.radius[circleIdx];

        // First, check if the circle overlaps with the tile
        if (!circleInBoxConservative(p.x, p.y, rad, tileL, tileR, tileT, tileB)) {
            continue; // Skip this circle if it doesn't overlap with the tile
        }

        // Now check if the circle overlaps with this specific pixel
        if (!circleInBoxConservative(p.x, p.y, rad, boxL, boxR, boxT, boxB)) {
            continue; // Skip this circle if it doesn't overlap with the pixel
        }

        // Compute the distance from pixel center to circle center
        float diffX = p.x - pixelCenterNorm.x;
        float diffY = p.y - pixelCenterNorm.y;
        float pixelDist = diffX * diffX + diffY * diffY;
        float maxDist = rad * rad;

        // If pixel is within the circle, update its color
        if (pixelDist <= maxDist) {
            // Compute the color of the circle at this pixel
            float3 rgb;
            float alpha;

            if (cuConstRendererParams.sceneName == SNOWFLAKES || 
                cuConstRendererParams.sceneName == SNOWFLAKES_SINGLE_FRAME) {
                const float kCircleMaxAlpha = .5f;
                const float falloffScale = 4.f;

                float normPixelDist = sqrt(pixelDist) / rad;
                rgb = lookupColor(normPixelDist);

                float maxAlpha = .6f + .4f * (1.f-p.z);
                maxAlpha = kCircleMaxAlpha * fmaxf(fminf(maxAlpha, 1.f), 0.f);
                alpha = maxAlpha * exp(-1.f * falloffScale * normPixelDist * normPixelDist);
            } else {
                // simple: each circle has an assigned color
                rgb = *(float3*)&(cuConstRendererParams.color[index3]);
                alpha = .5f;
            }

            float oneMinusAlpha = 1.f - alpha;

            // Update the local accumulator with this circle's contribution
            pixelColor.x = alpha * rgb.x + oneMinusAlpha * pixelColor.x;
            pixelColor.y = alpha * rgb.y + oneMinusAlpha * pixelColor.y;
            pixelColor.z = alpha * rgb.z + oneMinusAlpha * pixelColor.z;
            pixelColor.w = alpha + pixelColor.w;
        }
    }

    // Write the final color back to global memory (only once per pixel)
    *imgPtr = pixelColor;
}

void
CudaRenderer::render() {
    // Set up a 2D grid of threads, where each thread processes one pixel
    dim3 blockDim(16, 16);
    dim3 gridDim(
        (image->width + blockDim.x - 1) / blockDim.x,
        (image->height + blockDim.y - 1) / blockDim.y);

    kernelRenderCirclesTiled<<<gridDim, blockDim>>>();
    cudaDeviceSynchronize();
}
