struct GlobalConstants {

    SceneName sceneName;

    int numCircles;
    float* position;
    float* velocity;
    float* color;
    float* radius;

    int imageWidth;
    int imageHeight;
    float* imageData;
};

// Global variable that is in scope, but read-only, for all cuda
// kernels.  The __constant__ modifier designates this variable will
// be stored in special "constant" memory on the GPU.
__constant__ GlobalConstants cuConstRendererParams;

// read-only lookup tables used to quickly compute noise (needed by
// advanceAnimation for the snowflake scene)
__constant__ int    cuConstNoiseYPermutationTable[256];
__constant__ int    cuConstNoiseXPermutationTable[256];
__constant__ float  cuConstNoise1DValueTable[256];

// color ramp table needed for the color ramp lookup shader
#define COLOR_MAP_SIZE 5
__constant__ float  cuConstColorRamp[COLOR_MAP_SIZE][3];

// including parts of the CUDA code from external files
#include "noiseCuda.cu_inl"

// kernelClearImageSnowflake -- (CUDA device code)
// Clears image to a gradient for snowflake scenes
__global__ void kernelClearImageSnowflake() {

    int imageX = blockIdx.x * blockDim.x + threadIdx.x;
    int imageY = blockIdx.y * blockDim.y + threadIdx.y;

    int width = cuConstRendererParams.imageWidth;
    int height = cuConstRendererParams.imageHeight;

    if (imageX >= width || imageY >= height)
        return;

    int offset = 4 * (imageY * width + imageX);
    float shade = .4f + .45f * static_cast<float>(height-imageY) / height;
    float4 value = make_float4(shade, shade, shade, 1.f);
    // write to global memory using float4 for efficiency
    *(float4*)(&cuConstRendererParams.imageData[offset]) = value;
}


// kernelClearImage --  (CUDA device code)
// Clears image to a solid color
__global__ void kernelClearImage(float r, float g, float b, float a) {

    int imageX = blockIdx.x * blockDim.x + threadIdx.x;
    int imageY = blockIdx.y * blockDim.y + threadIdx.y;

    int width = cuConstRendererParams.imageWidth;
    int height = cuConstRendererParams.imageHeight;

    if (imageX >= width || imageY >= height)
        return;

    int offset = 4 * (imageY * width + imageX);
    float4 value = make_float4(r, g, b, a);

    // write to global memory using float4 for efficiency
    *(float4*)(&cuConstRendererParams.imageData[offset]) = value;
}

// kernelAdvanceBouncingBalls, kernelAdvanceSnowflake without changes)

// kernelAdvanceFireWorks
__global__ void kernelAdvanceFireWorks() {
    const float dt = 1.f / 60.f;
    const float pi = 3.14159;
    const float maxDist = 0.25f;

    float* velocity = cuConstRendererParams.velocity;
    float* position = cuConstRendererParams.position;
    float* radius = cuConstRendererParams.radius;

    int index = blockIdx.x * blockDim.x + threadIdx.x;
    if (index >= cuConstRendererParams.numCircles)
        return;

    if (0 <= index && index < NUM_FIREWORKS) { // firework center; no update
        return;
    }
    // determine the fire-work center/spark indices
    int fIdx = (index - NUM_FIREWORKS) / NUM_SPARKS;
    int sfIdx = (index - NUM_FIREWORKS) % NUM_SPARKS;

    int index3i = 3 * fIdx;
    int sIdx = NUM_FIREWORKS + fIdx * NUM_SPARKS + sfIdx;
    int index3j = 3 * sIdx;

    float cx = position[index3i];
    float cy = position[index3i+1];

    // update position
    position[index3j] += velocity[index3j] * dt;
    position[index3j+1] += velocity[index3j+1] * dt;

    // fire-work sparks
    float sx = position[index3j];
    float sy = position[index3j+1];

    // compute vector from firework-spark
    float cxsx = sx - cx;
    float cysy = sy - cy;

    // compute distance from fire-work
    float dist = sqrt(cxsx * cxsx + cysy * cysy);
    if (dist > maxDist) { // restore to starting position
        // random starting position on fire-work's rim
        float angle = (sfIdx * 2 * pi)/NUM_SPARKS;
        float sinA = sin(angle);
        float cosA = cos(angle);
        float x = cosA * radius[fIdx];
        float y = sinA * radius[fIdx];

        position[index3j] = position[index3i] + x;
        position[index3j+1] = position[index3i+1] + y;
        position[index3j+2] = 0.0f;

        // travel scaled unit length
        velocity[index3j] = cosA/5.0;
        velocity[index3j+1] = sinA/5.0;
        velocity[index3j+2] = 0.0f;
    }
}

// kernelAdvanceHypnosis
__global__ void kernelAdvanceHypnosis() {
    int index = blockIdx.x * blockDim.x + threadIdx.x;
    if (index >= cuConstRendererParams.numCircles)
        return;

    float* radius = cuConstRendererParams.radius;

    float cutOff = 0.5f;
    // place circle back in center after reaching threshold radisus
    if (radius[index] > cutOff) {
        radius[index] = 0.02f;
    } else {
        radius[index] += 0.01f;
    }
}


// kernelAdvanceBouncingBalls
__global__ void kernelAdvanceBouncingBalls() {
    const float dt = 1.f / 60.f;
    const float kGravity = -2.8f; // sorry Newton
    const float kDragCoeff = -0.8f;
    const float epsilon = 0.001f;

    int index = blockIdx.x * blockDim.x + threadIdx.x;

    if (index >= cuConstRendererParams.numCircles)
        return;

    float* velocity = cuConstRendererParams.velocity;
    float* position = cuConstRendererParams.position;

    int index3 = 3 * index;
    // reverse velocity if center position < 0
    float oldVelocity = velocity[index3+1];
    float oldPosition = position[index3+1];

    if (oldVelocity == 0.f && oldPosition == 0.f) { // stop-condition
        return;
    }

    if (position[index3+1] < 0 && oldVelocity < 0.f) { // bounce ball
        velocity[index3+1] *= kDragCoeff;
    }

    // update velocity: v = u + at (only along y-axis)
    velocity[index3+1] += kGravity * dt;

    // update positions (only along y-axis)
    position[index3+1] += velocity[index3+1] * dt;

    if (fabsf(velocity[index3+1] - oldVelocity) < epsilon
        && oldPosition < 0.0f
        && fabsf(position[index3+1]-oldPosition) < epsilon) { // stop ball
        velocity[index3+1] = 0.f;
        position[index3+1] = 0.f;
    }
}

// kernelAdvanceSnowflake
__global__ void kernelAdvanceSnowflake() {

    int index = blockIdx.x * blockDim.x + threadIdx.x;

    if (index >= cuConstRendererParams.numCircles)
        return;

    const float dt = 1.f / 60.f;
    const float kGravity = -1.8f; // sorry Newton
    const float kDragCoeff = 2.f;

    int index3 = 3 * index;

    float* positionPtr = &cuConstRendererParams.position[index3];
    float* velocityPtr = &cuConstRendererParams.velocity[index3];

    // loads from global memory
    float3 position = *((float3*)positionPtr);
    float3 velocity = *((float3*)velocityPtr);

    // hack to make farther circles move more slowly
    float forceScaling = fmin(fmax(1.f - position.z, .1f), 1.f); // clamp

    // add some noise to the motion
    float3 noiseInput;
    noiseInput.x = 10.f * position.x;
    noiseInput.y = 10.f * position.y;
    noiseInput.z = 255.f * position.z;
    float2 noiseForce = cudaVec2CellNoise(noiseInput, index);
    noiseForce.x *= 7.5f;
    noiseForce.y *= 5.f;

    // drag
    float2 dragForce;
    dragForce.x = -1.f * kDragCoeff * velocity.x;
    dragForce.y = -1.f * kDragCoeff * velocity.y;

    // update positions
    position.x += velocity.x * dt;
    position.y += velocity.y * dt;

    // update velocities
    velocity.x += forceScaling * (noiseForce.x + dragForce.y) * dt;
    velocity.y += forceScaling * (kGravity + noiseForce.y + dragForce.y) * dt;

    float radius = cuConstRendererParams.radius[index];

    // if the snowflake has moved off screen, reset its position
    if ( (position.y + radius < 0.f) ||
         (position.x + radius) < -0.f ||
         (position.x - radius) > 1.f)
    {
        noiseInput.x = 255.f * position.x;
        noiseInput.y = 255.f * position.y;
        noiseInput.z = 255.f * position.z;
        noiseForce = cudaVec2CellNoise(noiseInput, index);

        position.x = .5f + .5f * noiseForce.x;
        position.y = 1.35f + radius;

        // restart from 0 vertical velocity.
        velocity.x = 2.f * noiseForce.y;
        velocity.y = 0.f;
    }

    // store updated positions and velocities to global memory
    *((float3*)positionPtr) = position;
    *((float3*)velocityPtr) = velocity;
}


// shadePixel -- (CUDA device code)
//
// Given a pixel center, circle properties, and a pointer to the pixel's data in
// global memory, calculates the circle's contribution and blends it *in place*.
// Ensures atomicity and order implicitly because it's called by only one thread
// per pixel, sequentially for each overlapping circle relevant to that pixel.
__device__ __inline__ void
shadePixel(int circleIndex, float2 pixelCenter, float3 p, float4* imagePtr) {

    // Calculate squared distance from pixel center to circle center
    // Note: The caller (kernelRenderCirclesTiled) already checked if
    // pixelDistSq <= maxDistSq before calling this function.
    // Recomputing pixelDistSq here is slightly redundant but keeps the function self-contained.
    // For maximum performance, the caller could compute sqrt(pixelDistSq)/rad and pass it.
    float diffX = p.x - pixelCenter.x;
    float diffY = p.y - pixelCenter.y;
    float pixelDistSq = diffX * diffX + diffY * diffY;

    float rad = cuConstRendererParams.radius[circleIndex];
    float maxDistSq = rad * rad;

    // Early exit if somehow called for a pixel outside the circle
    // (Should not happen with the current calling logic, but safe to keep).
    if (pixelDistSq > maxDistSq)
        return;

    float3 rgb;
    float alpha;

    // Determine circle color and alpha based on scene type
    if (cuConstRendererParams.sceneName == SNOWFLAKES || cuConstRendererParams.sceneName == SNOWFLAKES_SINGLE_FRAME) {
        const float kCircleMaxAlpha = .5f;
        const float falloffScale = 4.f;

        // Calculate normalized distance for shading effects
        float normPixelDist = sqrtf(pixelDistSq) / rad;
        rgb = lookupColor(normPixelDist); // Color lookup based on normalized distance

        // Calculate alpha based on depth (z) and distance from center
        float maxAlpha = .6f + .4f * (1.f-p.z);
        maxAlpha = kCircleMaxAlpha * fmaxf(fminf(maxAlpha, 1.f), 0.f); // Clamp and scale alpha
        alpha = maxAlpha * expf(-1.f * falloffScale * normPixelDist * normPixelDist); // Exponential falloff

    } else {
        // Simple case: fixed color per circle
        int index3 = 3 * circleIndex;
        // Use float3 load for color
        rgb = *(float3*)&(cuConstRendererParams.color[index3]);
        alpha = .5f; // Constant alpha for non-snowflake scenes
    }

    float oneMinusAlpha = 1.f - alpha;

    // ----- Blending Operation (Read-Modify-Write) -----
    // Read the current pixel color from global memory (*imagePtr)
    float4 existingColor = *imagePtr;

    // Blend the new circle's color over the existing color using standard "over" operator
    float4 newColor;
    newColor.x = alpha * rgb.x + oneMinusAlpha * existingColor.x;
    newColor.y = alpha * rgb.y + oneMinusAlpha * existingColor.y;
    newColor.z = alpha * rgb.z + oneMinusAlpha * existingColor.z;
    // The alpha channel update in the original code `newColor.w = alpha + existingColor.w;`
    // doesn't correspond to standard alpha blending. We'll keep it to match the reference behavior.
    // For standard alpha blending, it would be:
    // newColor.w = alpha + existingColor.w * oneMinusAlpha; // or simply existingColor.w if background is opaque
    newColor.w = alpha + existingColor.w; // Matching original reference behavior

    // Write the updated color back to the *same* global memory location
    *imagePtr = newColor;
    // ----- End Blending Operation -----
}


// kernelRenderCirclesTiled -- (CUDA device code) - OPTIMIZED VERSION
//
// Each thread renders a single pixel. It iterates through all circles,
// but quickly skips circles whose bounding boxes do not overlap the thread block's tile
// using circleInBoxConservative. This preserves the order and atomicity guarantees
// while significantly reducing computation.
__global__ void kernelRenderCirclesTiled(int tileWidth, int tileHeight) {
    // Calculate global pixel coordinates (x, y) for this thread
    int pixelX = blockIdx.x * blockDim.x + threadIdx.x;
    int pixelY = blockIdx.y * blockDim.y + threadIdx.y;

    int imageWidth = cuConstRendererParams.imageWidth;
    int imageHeight = cuConstRendererParams.imageHeight;

    // Boundary check: Exit if thread's pixel is outside the image
    if (pixelX >= imageWidth || pixelY >= imageHeight)
        return;

    // Precompute inverse width/height for normalization
    float invWidth = 1.f / imageWidth;
    float invHeight = 1.f / imageHeight;

    // Calculate the center of this thread's pixel in normalized coordinates [0, 1]
    float2 pixelCenterNorm = make_float2(invWidth * (static_cast<float>(pixelX) + 0.5f),
                                        invHeight * (static_cast<float>(pixelY) + 0.5f));

    // Calculate the bounding box of this thread block's tile in normalized coordinates [0, 1]
    // Tile boundaries correspond to the edges of the pixels in the tile.
    float tileBoxL = invWidth * (blockIdx.x * blockDim.x);           // Left edge of the leftmost pixel
    float tileBoxR = invWidth * ((blockIdx.x * blockDim.x) + blockDim.x); // Right edge of the rightmost pixel
    float tileBoxB = invHeight * (blockIdx.y * blockDim.y);           // Bottom edge of the bottommost pixel
    float tileBoxT = invHeight * ((blockIdx.y * blockDim.y) + blockDim.y); // Top edge of the topmost pixel

    // Get a pointer to this pixel's color data (RGBA float4) in global memory
    float4* imgPtr = (float4*)(&cuConstRendererParams.imageData[4 * (pixelY * imageWidth + pixelX)]);

    // Iterate through all circles in the scene, processing them in order
    for (int circleIndex = 0; circleIndex < cuConstRendererParams.numCircles; circleIndex++) {
        int index3 = 3 * circleIndex;

        // Read circle position (center x, y, z) and radius from global memory
        float3 p = *(float3*)(&cuConstRendererParams.position[index3]);
        float rad = cuConstRendererParams.radius[circleIndex];

        // ***** OPTIMIZATION: Tile Culling *****
        // Use the conservative circle-in-box test to check if the circle
        // potentially overlaps with this thread block's tile.
        if (circleInBoxConservative(p.x, p.y, rad, tileBoxL, tileBoxR, tileBoxT, tileBoxB))
        {
            // The circle *might* overlap the tile.
            // Now perform the precise check for this specific pixel.

            // Compute squared distance from pixel center to circle center
            float diffX = p.x - pixelCenterNorm.x;
            float diffY = p.y - pixelCenterNorm.y;
            float pixelDistSq = diffX * diffX + diffY * diffY;
            float maxDistSq = rad * rad;

            // If the pixel center is within the circle's radius (precise check)
            if (pixelDistSq <= maxDistSq) {
                // Blend this circle's contribution onto the pixel's current color.
                // shadePixel reads from *imgPtr, blends, and writes back to *imgPtr.
                shadePixel(circleIndex, pixelCenterNorm, p, imgPtr);
            }
        }
        // If circleInBoxConservative returned false, the circle is definitely
        // outside this tile, so we skip the precise check and shading for this circle.
    }
    // After the loop, *imgPtr contains the final, correctly ordered blended color for this pixel.
}

CudaRenderer::CudaRenderer() {
    image = NULL;
    numCircles = 0;
    position = NULL; // Host pointers
    velocity = NULL;
    color = NULL;
    radius = NULL;
    cudaDevicePosition = NULL; // Device pointers
    cudaDeviceVelocity = NULL;
    cudaDeviceColor = NULL;
    cudaDeviceRadius = NULL;
    cudaDeviceImageData = NULL;
}

CudaRenderer::~CudaRenderer() {
    if (image) {
        delete image; // Free host image buffer
    }

    // Host memory (position, velocity, etc.) is assumed to be managed
    // externally (e.g., by SceneLoader) based on typical assignment structure.
    // If CudaRenderer allocated it, it should free it here.

    // Free CUDA device memory
    if (cudaDevicePosition) cudaCheckError(cudaFree(cudaDevicePosition));
    if (cudaDeviceVelocity) cudaCheckError(cudaFree(cudaDeviceVelocity));
    if (cudaDeviceColor) cudaCheckError(cudaFree(cudaDeviceColor));
    if (cudaDeviceRadius) cudaCheckError(cudaFree(cudaDeviceRadius));
    if (cudaDeviceImageData) cudaCheckError(cudaFree(cudaDeviceImageData));
}

const Image*
CudaRenderer::getImage() {
    // Copy the final rendered image from device memory back to host memory
    printf("Copying image data from device to host...\n");
    cudaCheckError(cudaMemcpy(image->data,
               cudaDeviceImageData,
               sizeof(float) * 4 * image->width * image->height,
               cudaMemcpyDeviceToHost));
    // Ensure the copy completes before returning the host pointer
    cudaCheckError(cudaDeviceSynchronize());
    printf("Image data copied.\n");
    return image;
}

void
CudaRenderer::loadScene(SceneName scene) {
    sceneName = scene;
    // loadCircleScene is expected to populate the host pointers:
    // numCircles, position, velocity, color, radius
    loadCircleScene(sceneName, numCircles, position, velocity, color, radius);
    printf("Loaded scene %d with %d circles.\n", (int)sceneName, numCircles);
}

void
CudaRenderer::setup() {
    // Initialize CUDA, print device info
    int deviceCount = 0;
    cudaCheckError(cudaGetDeviceCount(&deviceCount));
    printf("---------------------------------------------------------\\n");
    printf("Initializing CUDA for CudaRenderer\n");
    printf("Found %d CUDA devices\n", deviceCount);
    for (int i=0; i<deviceCount; i++) {
        cudaDeviceProp deviceProps;
        cudaCheckError(cudaGetDeviceProperties(&deviceProps, i));
        printf("Device %d: %s\n", i, deviceProps.name);
        printf("   SMs:        %d\n", deviceProps.multiProcessorCount);
        printf("   Global mem: %.0f MB\n", static_cast<float>(deviceProps.totalGlobalMem) / (1024 * 1024));
        printf("   CUDA Cap:   %d.%d\n", deviceProps.major, deviceProps.minor);
    }
    printf("---------------------------------------------------------\\n");

    // Allocate memory on the CUDA device for scene data and image buffer
    printf("Allocating device memory...\n");
    cudaCheckError(cudaMalloc(&cudaDevicePosition, sizeof(float) * 3 * numCircles));
    cudaCheckError(cudaMalloc(&cudaDeviceVelocity, sizeof(float) * 3 * numCircles));
    cudaCheckError(cudaMalloc(&cudaDeviceColor, sizeof(float) * 3 * numCircles));
    cudaCheckError(cudaMalloc(&cudaDeviceRadius, sizeof(float) * numCircles));
    // Ensure image dimensions are set via allocOutputImage before setup is called
    if (!image) {
        fprintf(stderr, "Error: allocOutputImage must be called before setup().\n");
        exit(1);
    }
    cudaCheckError(cudaMalloc(&cudaDeviceImageData, sizeof(float) * 4 * image->width * image->height));
    printf("Device memory allocated.\n");


    // Copy scene data from host arrays to device arrays
    printf("Copying scene data to device...\n");
    cudaCheckError(cudaMemcpy(cudaDevicePosition, position, sizeof(float) * 3 * numCircles, cudaMemcpyHostToDevice));
    cudaCheckError(cudaMemcpy(cudaDeviceVelocity, velocity, sizeof(float) * 3 * numCircles, cudaMemcpyHostToDevice));
    cudaCheckError(cudaMemcpy(cudaDeviceColor, color, sizeof(float) * 3 * numCircles, cudaMemcpyHostToDevice));
    cudaCheckError(cudaMemcpy(cudaDeviceRadius, radius, sizeof(float) * numCircles, cudaMemcpyHostToDevice));
    printf("Scene data copied.\n");


    // Initialize parameters structure and copy it to constant memory
    printf("Setting up constant memory parameters...\n");
    GlobalConstants params;
    params.sceneName = sceneName;
    params.numCircles = numCircles;
    params.imageWidth = image->width;
    params.imageHeight = image->height;
    params.position = cudaDevicePosition;
    params.velocity = cudaDeviceVelocity;
    params.color = cudaDeviceColor;
    params.radius = cudaDeviceRadius;
    params.imageData = cudaDeviceImageData;
    cudaCheckError(cudaMemcpyToSymbol(cuConstRendererParams, &params, sizeof(GlobalConstants)));

    // Copy noise tables to constant memory
    int* permX;
    int* permY;
    float* value1D;
    getNoiseTables(&permX, &permY, &value1D);
    cudaCheckError(cudaMemcpyToSymbol(cuConstNoiseXPermutationTable, permX, sizeof(int) * 256));
    cudaCheckError(cudaMemcpyToSymbol(cuConstNoiseYPermutationTable, permY, sizeof(int) * 256));
    cudaCheckError(cudaMemcpyToSymbol(cuConstNoise1DValueTable, value1D, sizeof(float) * 256));

    // Copy color ramp table to constant memory
    float lookupTable[COLOR_MAP_SIZE][3] = {
        {1.f, 1.f, 1.f}, {1.f, 1.f, 1.f}, {.8f, .9f, 1.f},
        {.8f, .9f, 1.f}, {.8f, 0.8f, 1.f}
    };
    cudaCheckError(cudaMemcpyToSymbol(cuConstColorRamp, lookupTable, sizeof(float) * 3 * COLOR_MAP_SIZE));

    printf("Constant memory setup complete.\n");
    // Ensure all setup operations (allocs, copies) are complete
    cudaCheckError(cudaDeviceSynchronize());
}

void
CudaRenderer::allocOutputImage(int width, int height) {
    // Allocate host image buffer
    if (image) delete image;
    image = new Image(width, height);
    printf("Allocated host output image buffer (%d x %d).\n", width, height);
    // Note: Corresponding device memory (cudaDeviceImageData) is allocated in setup().
}

void
CudaRenderer::clearImage() {
    // Use 2D thread blocks, e.g., 16x16 = 256 threads
    const int TILE_WIDTH = 16;
    const int TILE_HEIGHT = 16;
    dim3 blockDim(TILE_WIDTH, TILE_HEIGHT, 1);
    // Calculate grid dimensions to cover the entire image
    dim3 gridDim(
        (image->width + blockDim.x - 1) / blockDim.x,
        (image->height + blockDim.y - 1) / blockDim.y);

    printf("Clearing image on device (Grid: %d x %d, Block: %d x %d)...\n", gridDim.x, gridDim.y, blockDim.x, blockDim.y);

    // Launch the appropriate clear kernel based on scene
    if (sceneName == SNOWFLAKES || sceneName == SNOWFLAKES_SINGLE_FRAME) {
        kernelClearImageSnowflake<<<gridDim, blockDim>>>();
    } else {
        // Clear to white (1,1,1,1) for other scenes
        kernelClearImage<<<gridDim, blockDim>>>(1.f, 1.f, 1.f, 1.f);
    }
    // Check for errors during kernel launch
    cudaCheckError(cudaGetLastError());
    // Ensure clear operation is complete before proceeding
    cudaCheckError(cudaDeviceSynchronize());
    printf("Image cleared.\n");
}

void
CudaRenderer::advanceAnimation() {
     // Use 1D blocks for animation kernels (processing circles)
    const int ANIM_BLOCK_SIZE = 256; // Or 512, depending on kernel needs/performance
    dim3 blockDim(ANIM_BLOCK_SIZE, 1);
    // Calculate grid dimension to cover all circles
    dim3 gridDim((numCircles + blockDim.x - 1) / blockDim.x);

    // Select and launch the appropriate animation kernel
    bool animation_launched = false;
    if (sceneName == SNOWFLAKES) {
        printf("Advancing snowflake animation...\n");
        kernelAdvanceSnowflake<<<gridDim, blockDim>>>();
        animation_launched = true;
    } else if (sceneName == BOUNCING_BALLS) {
        printf("Advancing bouncing balls animation...\n");
        kernelAdvanceBouncingBalls<<<gridDim, blockDim>>>();
        animation_launched = true;
    } else if (sceneName == HYPNOSIS) {
        printf("Advancing hypnosis animation...\n");
        kernelAdvanceHypnosis<<<gridDim, blockDim>>>();
        animation_launched = true;
    } else if (sceneName == FIREWORKS) {
        printf("Advancing fireworks animation...\n");
        kernelAdvanceFireWorks<<<gridDim, blockDim>>>();
        animation_launched = true;
    }

    if (animation_launched) {
        // Check for kernel launch errors
        cudaCheckError(cudaGetLastError());
         // Ensure animation update is finished before rendering
        cudaCheckError(cudaDeviceSynchronize());
        printf("Animation advanced.\n");
    } else {
        // No animation needed for this scene
    }
}


void
CudaRenderer::render() {
    // Define tile dimensions (match the block dimensions)
    // 16x16 = 256 threads per block is generally a good starting point
    const int TILE_WIDTH = 16;
    const int TILE_HEIGHT = 16;
    dim3 blockDim(TILE_WIDTH, TILE_HEIGHT);

    // Calculate grid dimensions needed to cover all pixels with these tiles
    dim3 gridDim(
        (image->width + blockDim.x - 1) / blockDim.x,
        (image->height + blockDim.y - 1) / blockDim.y);

    printf("Rendering image with Tiled Kernel (Grid: %d x %d, Block: %d x %d)...\n", gridDim.x, gridDim.y, blockDim.x, blockDim.y);

    // Launch the optimized tiled rendering kernel, passing tile dimensions
    kernelRenderCirclesTiled<<<gridDim, blockDim>>>(TILE_WIDTH, TILE_HEIGHT);

    // Check for kernel launch errors immediately
    cudaCheckError(cudaGetLastError());

    // Synchronize to ensure rendering is complete before the function returns
    // (e.g., before getImage() might be called for timing or verification)
    cudaCheckError(cudaDeviceSynchronize());
    printf("Rendering complete.\n");
}
